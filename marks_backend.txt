@app.route('/api/teacher/students_v2/<int:eid>/<int:sid>')
@login_required
def students_v2_new(eid, sid):
    u = session['user']
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    # Humne DISTINCT aur GROUP BY laga diya hai taake duplicates ka sawal hi paida na ho
    query = '''
        SELECT DISTINCT 
            s.id, 
            s.full_name, 
            s.roll_number, 
            m.obtained_marks, 
            m.remarks
        FROM apsokara_student s
        LEFT JOIN (
            SELECT student_id, exam_id, sub_id, obtained_marks, remarks 
            FROM apsokara_marks 
            WHERE exam_id = ? AND sub_id = ?
            GROUP BY student_id
        ) m ON s.id = m.student_id
        WHERE s.student_class = ? AND s.student_section = ? AND s.wing = ?
        GROUP BY s.id
        ORDER BY CAST(s.roll_number AS INTEGER) ASC
    '''
    students = conn.execute(query, (eid, sid, u['assigned_class'], u['assigned_section'], u['wing'])).fetchall()
    conn.close()
    return jsonify({'students': [dict(s) for s in students]})
